# --- состояние до изменений ---
cd /home/user/docs
ls -l
# --- touch: новый файл, несколько файлов, существующий файл ---
touch new.txt
touch a.txt b.txt
touch notes.txt
ls -l
head new.txt
touch /home/user/music/song.txt
ls /home/user/music
# --- rm: файлы ---
rm a.txt
rm b.txt new.txt

# --- rm: папки (нужен -r), ключ -f ---
rm work
rm -r work
ls
rm -rf /home/user/music
ls /home/user
